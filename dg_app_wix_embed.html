// ============================================================ // DG MASTER PT LIST — BACKEND // File: backend/dgApp.web.js // ============================================================ import wixData from 'wix-data'; import { webMethod, Permissions } from 'wix-web-module'; import { secrets } from 'wix-secrets-backend.v2'; import { elevate } from 'wix-auth'; import baseRosterSeed from './dgBaseRosterSeed.js'; // ============================================================ // COLLECTIONS // ============================================================ const COLLECTIONS = { status: 'Import1', duplicates: 'Import2', edited: 'Import3', deleted: 'Import4', completion: 'Import5', imported: 'Import6', importLog: 'Import7', changeLog: 'Import8', baseRoster: 'Import9', sessions: 'DGTeamSessions' }; // ============================================================ // SETTINGS // ============================================================ const MAX_LOG_FETCH = 1000; // Session duration = 8 hours const SESSION_DURATION_MS = 8 * 60 * 60 * 1000; // ============================================================ // ELEVATED SECRET ACCESS // ============================================================ const getSecretValue = elevate(secrets.getSecretValue); // ============================================================ // BASIC HELPERS // ============================================================ function toJsonText(value) { return ( value === null || value === undefined ) ? '' : JSON.stringify(value); } function fromJsonText( text, fallback ) { if (!text) { return fallback; } try { return JSON.parse(text); } catch (e) { return fallback; } } // ============================================================ // SAFE ID // ============================================================ function safeId(raw) { const s = String(raw); let h = 0; for ( let i = 0; i < s.length; i++ ) { h = ( Math.imul( 31, h ) + s.charCodeAt(i) ) | 0; } const hex = (h >>> 0).toString(16); const slug = s .replace( /[^A-Za-z0-9]+/g, '_' ) .replace( /^_+|_+$/g, '' ) .slice(0, 60); return ( `${slug || 'k'}_${hex}` ); } // ============================================================ // SESSION TOKEN // ============================================================ function createSessionToken() { const part1 = Date.now().toString(36); const part2 = Math.random() .toString(36) .substring(2, 14); const part3 = Math.random() .toString(36) .substring(2, 14); const part4 = Math.random() .toString(36) .substring(2, 14); return [ part1, part2, part3, part4 ].join('-'); } // ============================================================ // GET TEAM PIN // ============================================================ async function getTeamPin() { const response = await getSecretValue( 'DG_TEAM_PIN' ); const pin = response?.value ?? response?.secret ?? response?.text ?? ''; if (!pin) { throw new Error( 'DG_TEAM_PIN secret is empty or not configured.' ); } return String(pin).trim(); } // ============================================================ // VALIDATE ACCESS TOKEN // ============================================================ async function validateAccessToken( accessToken ) { if (!accessToken) { return false; } try { const result = await wixData .query( COLLECTIONS.sessions ) .eq( 'token', String(accessToken) ) .limit(1) .find({ suppressAuth: true }); if (!result.items.length) { return false; } const session = result.items[0]; if (!session.expiresAt) { return false; } const expires = new Date( session.expiresAt ).getTime(); if ( !Number.isFinite(expires) || expires <= Date.now() ) { // Remove expired session try { await wixData.remove( COLLECTIONS.sessions, session._id, { suppressAuth: true } ); } catch (e) { console.warn( 'Could not remove expired session.', e ); } return false; } return true; } catch (error) { console.error( 'Access token validation error:', error ); return false; } } // ============================================================ // VERIFY TEAM PIN // ============================================================ export const verifyTeamPin = webMethod( Permissions.Anyone, async (pin) => { try { console.log( 'DG PIN: verification started' ); const submittedPin = pin === null || pin === undefined ? '' : String(pin).trim(); if (!submittedPin) { return { success: false, message: 'Please enter the Team PIN.' }; } // ---------------------------------------------- // Get secret from Wix Secrets Manager // ---------------------------------------------- console.log( 'DG PIN: getting secret' ); const correctPin = await getTeamPin(); // ---------------------------------------------- // Compare PIN // ---------------------------------------------- if ( submittedPin !== correctPin ) { console.log( 'DG PIN: incorrect' ); return { success: false, message: 'Incorrect Team PIN.' }; } console.log( 'DG PIN: correct' ); // ---------------------------------------------- // Create temporary session // ---------------------------------------------- const accessToken = createSessionToken(); const expiresAt = new Date( Date.now() + SESSION_DURATION_MS ); // ---------------------------------------------- // Save session // ---------------------------------------------- await wixData.insert( COLLECTIONS.sessions, { title: 'DG Team Session', token: accessToken, expiresAt: expiresAt }, { suppressAuth: true } ); console.log( 'DG PIN: session created' ); return { success: true, accessToken: accessToken, expiresIn: SESSION_DURATION_MS }; } catch (error) { console.error( 'DG PIN ERROR:', error ); return { success: false, message: 'Team PIN verification failed: ' + ( error?.message || String(error) ) }; } } ); // ============================================================ // FETCH ALL // ============================================================ async function fetchAll( collectionId, options ) { const desc = options && options.desc; let query = wixData .query(collectionId) .limit(1000); if (desc) { query = query.descending(desc); } const results = []; let page = await query.find({ suppressAuth: true }); results.push( ...page.items ); while ( page.hasNext() && results.length < ( desc ? MAX_LOG_FETCH : 100000 ) ) { page = await page.next(); results.push( ...page.items ); } return results; } // ============================================================ // FETCH BASE ROSTER // ============================================================ async function fetchBaseRoster() { let rows = []; try { rows = await fetchAll( COLLECTIONS.baseRoster ); } catch (e) { console.warn( 'DG app: base roster unavailable.' ); } const rawSheets = {}; for (const r of rows) { rawSheets[r.sheetId] = { sheetName: r.sheetName, headers: fromJsonText( r.headers, [] ), rowCount: r.rowCount, rows: fromJsonText( r.rows, [] ) }; } return rawSheets; } // ============================================================ // BUILD INCREMENTAL STATE // ============================================================ async function buildIncrementalState() { const [ statusRows, dupRows, editedRows, deletedRows, completionRows, importedRows, importLogRows, changeLogRows ] = await Promise.all([ fetchAll( COLLECTIONS.status ), fetchAll( COLLECTIONS.duplicates ), fetchAll( COLLECTIONS.edited ), fetchAll( COLLECTIONS.deleted ), fetchAll( COLLECTIONS.completion ), fetchAll( COLLECTIONS.imported ), fetchAll( COLLECTIONS.importLog, { desc: 'ts' } ), fetchAll( COLLECTIONS.changeLog, { desc: 'ts' } ) ]); const statusOverrides = {}; for ( const r of statusRows ) { statusOverrides[r.uid] = r.status; } const duplicateReviews = {}; for ( const r of dupRows ) { duplicateReviews[r.key] = { reviewed: r.reviewed, note: r.note, excluded: r.excluded, keepUid: r.keepUid }; } const editedRowsOut = {}; for ( const r of editedRows ) { editedRowsOut[r.uid] = fromJsonText( r.patch, {} ); } const deletedRowsOut = {}; for ( const r of deletedRows ) { deletedRowsOut[r.uid] = !!r.deleted; } const completionOverrides = {}; for ( const r of completionRows ) { completionOverrides[r.sheetId] = fromJsonText( r.config, null ); } const imported = {}; for ( const r of importedRows ) { if (!imported[r.sheetId]) { imported[r.sheetId] = []; } imported[r.sheetId][r.rowIndex] = fromJsonText( r.row, {} ); } for ( const sheetId of Object.keys(imported) ) { imported[sheetId] = imported[sheetId] .map( row => row || null ) .filter( row => row !== null ); } return { imported, statusOverrides, duplicateReviews, editedRows: editedRowsOut, deletedRows: deletedRowsOut, completionOverrides, importLog: importLogRows.map( stripSystemFields ), changeLog: changeLogRows.map( stripSystemFields ) }; } // ============================================================ // STRIP SYSTEM FIELDS // ============================================================ function stripSystemFields(item) { const { _id, _owner, _createdDate, _updatedDate, ...rest } = item; return rest; } // ============================================================ // INITIALIZE // ============================================================ export const dgAppInit = webMethod( Permissions.Anyone, async (accessToken) => { const authorized = await validateAccessToken( accessToken ); if (!authorized) { throw new Error( 'Unauthorized. Please enter the Team PIN.' ); } const [ state, rawSheets ] = await Promise.all([ buildIncrementalState(), fetchBaseRoster() ]); return { ...state, rawSheets }; } ); // ============================================================ // REFRESH // ============================================================ export const dgAppRefresh = webMethod( Permissions.Anyone, async (accessToken) => { const authorized = await validateAccessToken( accessToken ); if (!authorized) { throw new Error( 'Unauthorized. Please enter the Team PIN.' ); } return buildIncrementalState(); } ); // ============================================================ // SAVE // ============================================================ export const dgAppSave = webMethod( Permissions.Anyone, async ( accessToken, type, payload ) => { const authorized = await validateAccessToken( accessToken ); if (!authorized) { throw new Error( 'Unauthorized. Please enter the Team PIN.' ); } switch (type) { case 'statusOverride': return saveStatusOverride( payload ); case 'duplicateReview': return saveDuplicateReview( payload ); case 'editedRow': return saveEditedRow( payload ); case 'deletedRow': return saveDeletedRow( payload ); case 'completionOverride': return saveCompletionOverride( payload ); case 'importedRows': return saveImportedRows( payload ); case 'importLogEntry': return saveImportLogEntry( payload ); case 'changeLogEntry': return saveChangeLogEntry( payload ); default: throw new Error( 'dgAppSave: unknown type "' + type + '"' ); } } ); // ============================================================ // SAVE STATUS // ============================================================ async function saveStatusOverride( payload ) { const uid = payload.uid; const status = payload.status; return wixData.save( COLLECTIONS.status, { _id: safeId(uid), uid, status }, { suppressAuth: true } ); } // ============================================================ // SAVE DUPLICATE REVIEW // ============================================================ async function saveDuplicateReview( payload ) { const key = payload.key; const patch = payload.patch; return wixData.save( COLLECTIONS.duplicates, { _id: safeId(key), key, reviewed: !!( patch && patch.reviewed ), note: ( patch && patch.note ) || '', excluded: !!( patch && patch.excluded ), keepUid: ( patch && patch.keepUid ) || '' }, { suppressAuth: true } ); } // ============================================================ // SAVE EDITED ROW // ============================================================ async function saveEditedRow( payload ) { const uid = payload.uid; const patch = payload.patch; return wixData.save( COLLECTIONS.edited, { _id: safeId(uid), uid, patch: toJsonText(patch) }, { suppressAuth: true } ); } // ============================================================ // SAVE DELETED ROW // ============================================================ async function saveDeletedRow( payload ) { const uid = payload.uid; const deleted = payload.deleted; return wixData.save( COLLECTIONS.deleted, { _id: safeId(uid), uid, deleted: !!deleted }, { suppressAuth: true } ); } // ============================================================ // SAVE COMPLETION // ============================================================ async function saveCompletionOverride( payload ) { const sheetId = payload.sheetId; const config = payload.config; if ( config === null || config === undefined ) { try { await wixData.remove( COLLECTIONS.completion, sheetId, { suppressAuth: true } ); } catch (e) { if ( String( ( e && e.message ) || e ).indexOf( 'not found' ) === -1 ) { throw e; } } return { removed: true }; } return wixData.save( COLLECTIONS.completion, { _id: sheetId, sheetId, config: toJsonText(config) }, { suppressAuth: true } ); } // ============================================================ // SAVE IMPORTED ROWS // ============================================================ async function saveImportedRows( payload ) { const sheetId = payload.sheetId; const rows = payload.rows; if ( !rows || !rows.length ) { return { added: 0 }; } const existing = await wixData .query( COLLECTIONS.imported ) .eq( 'sheetId', sheetId ) .limit(1000) .count({ suppressAuth: true }); const items = rows.map( (row, i) => ({ _id: sheetId + '__imp' + (existing + i), sheetId, rowIndex: existing + i, row: toJsonText(row) }) ); await wixData.bulkInsert( COLLECTIONS.imported, items, { suppressAuth: true } ); return { added: items.length }; } // ============================================================ // SAVE IMPORT LOG // ============================================================ async function saveImportLogEntry( entry ) { return wixData.insert( COLLECTIONS.importLog, entry, { suppressAuth: true } ); } // ============================================================ // SAVE CHANGE LOG // ============================================================ async function saveChangeLogEntry( entry ) { return wixData.insert( COLLECTIONS.changeLog, entry, { suppressAuth: true } ); } // ============================================================ // SEED BASE ROSTER // ADMIN ONLY // ============================================================ export const seedBaseRoster = webMethod( Permissions.Admin, async () => { const items = Object.entries( baseRosterSeed ).map( ([sheetId, sheet]) => ({ _id: sheetId, sheetId, sheetName: sheet.sheetName, headers: toJsonText( sheet.headers ), rowCount: sheet.rowCount, rows: toJsonText( sheet.rows ) }) ); const CHUNK = 20; let saved = 0; for ( let i = 0; i < items.length; i += CHUNK ) { const chunk = items.slice( i, i + CHUNK ); await wixData.bulkSave( COLLECTIONS.baseRoster, chunk, { suppressAuth: true } ); saved += chunk.length; } return { sheetsSeeded: saved }; } );
